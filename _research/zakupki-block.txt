### demo-gov/index.html — обработчик

    /* ---------- Фильтр реестра закупок ----------
       Работает на месте; на боевом сайте выборку отдаёт CMS. */
    var yearBtn = t.closest("[data-year]");
    if (yearBtn) {
      var year = yearBtn.getAttribute("data-year");
      var rows = document.querySelectorAll("#registry tr");
      var shown = 0;
      for (var r = 0; r < rows.length; r++) {
        var fit = year === "all" || rows[r].getAttribute("data-y") === year;
        rows[r].hidden = !fit;
        if (fit) shown++;
      }
      press("[data-year]", yearBtn);
      document.getElementById("registryCount").textContent =
        (root.getAttribute("lang") === "kk" ? "Табылды: " : "Найдено записей: ") + shown;
      return;
    }
